E.2 — Authentication and Role-Based Access Control

Every protected route in the system passes through the following middleware. The three token verifiers cover the three user types (admin, customer, driver). Role enforcement happens at the middleware layer, not in the client, so a malicious user with a valid JWT still cannot reach endpoints outside their role.

Supports: Section 3.2.2 (Non-Functional Requirements — Security), Section 1.5 (Objectives)
middleware/authMiddleware.js
const adminService    = require('../services/AdminServices');
const customerService = require('../services/CustomerServices');
const driverService   = require('../services/DriverServices');

// ─────────────────────────────────────────
// ADMIN — verifies JWT and attaches req.admin
// ─────────────────────────────────────────
const verifyAdminToken = async (req, res, next) => {
  try {
    const token = req.headers.authorization?.split(' ')[1];
    if (!token) {
      return res.status(401).json({ success: false, message: 'Access denied. No token provided.' });
    }

    const response = await adminService.verifyToken(token);
    if (!response.success) {
      return res.status(401).json({ success: false, message: response.message });
    }

    req.admin = response.data;
    next();
  } catch (error) {
    return res.status(401).json({ success: false, message: 'Authentication failed' });
  }
};

// ─────────────────────────────────────────
// CUSTOMER — attaches req.customer
// ─────────────────────────────────────────
const verifyCustomerToken = async (req, res, next) => {
  try {
    const token = req.headers.authorization?.split(' ')[1];
    if (!token) return res.status(401).json({ success: false, message: 'Access denied.' });

    const response = await customerService.verifyToken(token);
    if (!response.success) return res.status(401).json({ success: false, message: response.message });

    req.customer = response.data;
    next();
  } catch (error) {
    return res.status(401).json({ success: false, message: 'Authentication failed' });
  }
};

// ─────────────────────────────────────────
// DRIVER — attaches req.user for controller compatibility
// ─────────────────────────────────────────
const verifyDriverToken = async (req, res, next) => {
  try {
    const token = req.headers.authorization?.split(' ')[1];
    if (!token) return res.status(401).json({ success: false, message: 'Access denied.' });

    const response = await driverService.verifyToken(token);
    if (!response.success) return res.status(401).json({ success: false, message: response.message });

    const driverData = response.data;
    req.driver = driverData;
    req.user = {
      id: driverData._id || driverData.id,
      driverId: driverData.driverId,
      email: driverData.email,
      role: 'driver',
      // ...additional driver fields
    };
    next();
  } catch (error) {
    return res.status(401).json({ success: false, message: 'Authentication failed' });
  }
};

// ─────────────────────────────────────────
// ROLE ENFORCEMENT — for admin-only sub-roles
// (e.g. only Super Admin can delete another admin)
// ─────────────────────────────────────────
const checkRole = (...roles) => {
  return (req, res, next) => {
    if (!req.admin) {
      return res.status(403).json({ success: false, message: 'Admins only.' });
    }
    if (!roles.includes(req.admin.role)) {
      return res.status(403).json({
        success: false,
        message: `Access denied. Required role: ${roles.join(' or ')}`,
      });
    }
    next();
  };
};

E.3 — Order Lifecycle and State Machine

The backend enforces a strict forward-only status flow. Any attempt to skip a step (e.g. Pending → Completed directly) is rejected with a descriptive error, regardless of what the client sends. This is a server-side rule; the client UI simply mirrors it.

Supports: Section 3.7.2 (Admin Activity Diagram — Process Transaction), Section 4.2 (Objective No. 2)
services/OrderServices.js — State Machine Constants
const VALID_STATUSES = [
  "Pending",
  "Confirmed",
  "Scheduled",
  "In Production",
  "Out for Delivery",
  "Completed",
  "Cancelled",
];

const STATUS_FLOW = [
  "Pending", "Confirmed", "Scheduled",
  "In Production", "Out for Delivery", "Completed",
];

// Translates the display-only "Ready to Pick-up" (used for Pick-up orders)
// into its canonical DB status "Out for Delivery".
function normalizeIncomingStatus(status) {
  if (status === "Ready to Pick-up") return "Out for Delivery";
  return status;
}

// ── THE RULE ────────────────────────────────────────────
// Forward only, one step at a time. "Cancelled" is always allowed
// as a terminal escape hatch from any non-terminal state.
// ───────────────────────────────────────────────────────
function isValidTransition(currentStatus, nextStatus) {
  if (nextStatus === "Cancelled")         return true;
  if (nextStatus === currentStatus)        return true;

  const currentIdx = STATUS_FLOW.indexOf(currentStatus);
  const nextIdx    = STATUS_FLOW.indexOf(nextStatus);
  if (currentIdx === -1 || nextIdx === -1) return false;

  return nextIdx === currentIdx + 1;
}
services/OrderServices.js — updateOrderStatus (guard clauses)
async updateOrderStatus(orderId, newStatus, notes, productionSchedule = null,
                       driverId = null, user = null, options = {}) {
  const { codCollected = false } = options;

  try {
    newStatus = normalizeIncomingStatus(newStatus);

    if (!VALID_STATUSES.includes(newStatus)) {
      return { success: false, message: "Invalid status" };
    }

    let order = await Order.findOne({ orderId });
    if (!order) return { success: false, message: "Order not found" };

    // Rule: cannot confirm an order until a downpayment is recorded.
    if (newStatus === 'Confirmed' && order.paymentStatus === 'Unpaid') {
      return { success: false, message:
        'Cannot confirm an order before a downpayment is verified.' };
    }

    // Rule: cannot complete without collecting remaining balance on Partial orders.
    if (newStatus === "Completed") {
      const needsCodCollection = order.paymentStatus === "Partial";
      if (needsCodCollection && codCollected !== true) {
        return { success: false, message:
          "Confirm remaining balance collected before completing." };
      }
      // ... payment settlement (see E.6)
    }

    // Rule: cannot act on terminal orders.
    if (order.status === "Completed" || order.status === "Cancelled") {
      return { success: false, message:
        `Cannot change status of a ${order.status.toLowerCase()} order` };
    }

// ── THE TRANSITION GUARD ─────────────────────────────
// Enforced on every single status update, regardless of caller
// (admin UI, driver app, or direct API access).
if (!isValidTransition(order.status, newStatus)) {
  return { success: false, message:
    `Cannot move order from "${order.status}" to "${newStatus}". ` +
    `Valid next step: "${STATUS_FLOW[STATUS_FLOW.indexOf(order.status) + 1] || "Cancelled"}".` };
}

    // ... remaining mutation + history push + save
  } catch (error) {
    throw error;
  }
}

E.4 — Transactional Order Creation

For company-product orders, stock decrement and order insert are wrapped in a single MongoDB transaction. If any product's stock is insufficient, the entire operation rolls back and no order is written. Total pricing is recomputed server-side; the client-submitted amount is ignored.

Supports: Section 4.1 (Objective No. 1 — Reduced errors), Section 4.5 (Objective No. 5 — Real-time stock)

E.5 — Negotiation-in-Chat Flow

The signature feature of the system. An admin can adjust an order's pricing in chat, send a quote, and the customer can accept or reject. On acceptance, the order is confirmed. This section shows the three critical functions: admin repricing, admin quoting, and customer acceptance.

Supports: Figure 3.9.23 (Negotiation Page), Section 4.3 (Objective No. 3 — Order tracking)
services/OrderServices.js — negotiateOrder (guards + quantity + recompute)
async negotiateOrder(orderId, updates, admin) {
  try {
    const order = await Order.findOne({ orderId });
    if (!order) return { success: false, message: "Order not found" };

    // ── Guard: only Pending + Unpaid orders may be negotiated
    if (order.status !== "Pending") {
      return { success: false, message:
        `Cannot negotiate an order with status "${order.status}".` };
    }
    if (order.paymentStatus !== "Unpaid") {
      return { success: false, message:
        `Cannot negotiate an order with payment status "${order.paymentStatus}".` };
    }

    const adminName = admin?.firstName
      ? `${admin.firstName} ${admin.lastName}`
      : admin?.email || "Admin";

    const historyEntries = [];

    // ── Editable: quantity (with stock delta)
    if (updates.quantity != null) {
      const newQty = Number(updates.quantity);
      if (isNaN(newQty) || newQty <= 0) {
        return { success: false, message: "Quantity must be positive" };
      }
      if (newQty !== order.quantity) {
        const oldQty = order.quantity;

        // Restore old qty → check availability → deduct new qty
        // (performed on the referenced Product document)
        // ...see full implementation for stock rollback logic

        order.quantity = newQty;
        if (order.items?.[0]) order.items[0].quantity = newQty;

        historyEntries.push({
          field: "quantity", oldValue: oldQty, newValue: newQty,
          updatedBy: adminName, notes: updates.notes || "",
        });
      }
    }

    // ── Editable: unitPrice, designFee, deliveryMethod, shippingFee
    // Each is guarded, validated, and logged into historyEntries.
    // (see manuscript version for full body)

    // ── Server-side total recomputation — the customer never sees
    //    a value that was not computed here.
    let productSubtotal = 0;
    if (!order.isProvided && order.items?.length) {
      productSubtotal = order.items.reduce((sum, it) => sum + (it.estimatedTotal || 0), 0);
    }
const newAmount = productSubtotal + (order.designFee || 0) + (order.shippingFee || 0);
order.amount      = newAmount;
order.totalAmount = newAmount;

    if (historyEntries.length > 0) {
      order.negotiationStatus = "in_progress";
      order.pricingHistory.push(...historyEntries);
      await order.save();
    }

    return { success: true, data: order, changes: historyEntries };
  } catch (error) { throw error; }
}

E.6 — Payment Ledger with Atomic Updates

Payments are recorded as an append-only ledger (partialPayments) rather than a single paid/unpaid flag. The customer's lifetime totalSpent is incremented atomically with MongoDB's $inc operator, avoiding a read-modify-write race when two payment events land in parallel.

Supports: Section 4.3 (payment breakdown), Section 4.6 (Evaluation — Reliability)
services/OrderServices.js — updatePaymentStatus (paid branch + auto-record)
async updatePaymentStatus(orderId, paymentStatus, amountPaid = null, user = null, partialPayments = null) {
  try {
    const order = await Order.findOne({ orderId });
    if (!order) return { success: false, message: "Order not found" };

    const paidBefore = this._computeTotalPaid(order);

    // ── Paid: auto-record any remaining balance so the ledger is complete
if (paymentStatus === "Paid") {
  const totalAmount = Number(order.amount) || Number(order.totalAmount) || 0;
  const alreadyPaid = Array.isArray(order.partialPayments)
    ? order.partialPayments.reduce((s, p) => s + (Number(p.amount) || 0), 0)
    : 0;
  const remainingBalance = Math.max(0, totalAmount - alreadyPaid);

  if (remainingBalance > 0) {
    order.partialPayments = order.partialPayments || [];
    order.partialPayments.push({
      amount: remainingBalance,
      referenceNumber: null,
      date: new Date(),
      updatedBy: user ? `${user.firstName} ${user.lastName}` : "System (auto-recorded on Paid)",
    });
  }
}

    order.paymentStatus = paymentStatus;
    if (paymentStatus === "Paid") {
      order.paymentDetails = order.paymentDetails || {};
      order.paymentDetails.paidAt = new Date();
    }
    await order.save();

    // Apply the delta to the customer's lifetime total (atomic)
    const paidAfter = this._computeTotalPaid(order);
    const delta = paidAfter - paidBefore;
    if (delta !== 0) await this._applyTotalSpentDelta(order, delta);

    return { success: true, data: order };
  } catch (error) { throw error; }
}
services/OrderServices.js — _applyTotalSpentDelta (atomic $inc)
async _applyTotalSpentDelta(order, delta) {
  if (!delta || isNaN(delta)) return;

  let customer = null;
  if (order.orderedBy) customer = await Customer.findById(order.orderedBy);
  if (!customer) customer = await Customer.findOne({ email: order.customerEmail.toLowerCase() });
  if (!customer) return;

// Atomic increment. Avoids the read-modify-write race that would
// otherwise lose updates when two payments land in parallel.
await Customer.updateOne(
  { _id: customer._id },
  { $inc: { totalSpent: delta }, $set: { updatedAt: new Date() } },
);
}

E.7 — Delay Tracking End-to-End

Delays are stored as an append-only history on the order, orthogonal to status. When an admin reports a delay, the customer receives an in-chat system message and a banner on their order detail page. The original ETA is preserved so the customer can see the change side-by-side.

Supports: Section 4.3 (Customer transparency), Figure 3.9.23 (Negotiation and delay in chat)
services/OrderServices.js — reportDelay
async reportDelay(orderId, payload, reporter, reporterType = 'admin') {
  const { category, reason, notes, newExpectedDelivery } = payload || {};
  if (!reason?.trim()) return { success: false, message: 'Delay reason required' };

  const order = await Order.findOne({ orderId });
  if (!order) return { success: false, message: 'Order not found' };

  // Resolve any currently-active delay before starting a new one
  (order.delayHistory || []).forEach((d) => {
    if (d.isDelayed) {
      d.isDelayed = false;
      d.resolvedAt = new Date();
      d.resolvedBy = reporter?.firstName ? `${reporter.firstName} ${reporter.lastName}` : 'Admin';
    }
  });

  // Preserve the original ETA across multiple delays
  const originalETA = order.delayHistory?.find((d) => d.originalExpectedDelivery)?.originalExpectedDelivery
    || order.expectedDelivery
    || null;

order.delayHistory.push({
  isDelayed:                true,
  category:                 category || 'other',
  reason:                   reason.trim(),
  notes:                    notes || '',
  originalExpectedDelivery: originalETA,
  newExpectedDelivery:      newExpectedDelivery ? new Date(newExpectedDelivery) : null,
  reportedBy:               reporterName,
  reportedAt:               new Date(),
});

  if (newExpectedDelivery) order.expectedDelivery = new Date(newExpectedDelivery);
  await order.save();

  // Push a system message to the linked conversation so the
  // customer sees the delay in chat — not just on the order page.
  await this._pushDelayNoticeToChat(order, reporterName, reporterId);

  return { success: true, data: order };
}
services/OrderServices.js — _pushDelayNoticeToChat
async _pushDelayNoticeToChat(order, adminName, adminId) {
  try {
    const conversation = await Conversation.findOne({ orderId: order.orderId });
    if (!conversation) return;

    const lastDelay = order.delayHistory[order.delayHistory.length - 1];

    const notice = new Message({
      messageId:      await generateId('MSG'),
      conversationId: conversation.conversationId,
      senderType:     'admin',
      senderId:       adminId,
      senderName:     adminName,
      content:        `⚠️ Order ${order.orderId} delayed — ${lastDelay.reason}`,
contentType:    'delay-notice',
delayNoticeData: {
  orderId:                   order.orderId,
  category:                  lastDelay.category,
  reason:                    lastDelay.reason,
  originalExpectedDelivery:  lastDelay.originalExpectedDelivery,
  newExpectedDelivery:       lastDelay.newExpectedDelivery,
  reportedAt:                lastDelay.reportedAt,
},
      isDeleted: false,
    });

    await notice.save();

    conversation.lastMessage = `⚠️ Delay notice: ${lastDelay.reason}`;
    conversation.customerUnreadCount += 1;
    await conversation.save();

    // Broadcast via socket so the customer's chat updates live
    const io = global.__io__;
    if (io) io.to(`conv_${conversation.conversationId}`).emit('new-message', notice.toObject());
  } catch (err) {
    // Never break the delay flow over a chat failure
  }
}

E.8 — Real-Time Broadcast Layer

Every client (admin, customer, driver) opens a single Socket.io connection on login. The server joins each connection to a set of rooms based on the JWT payload. Mutating services then broadcast to those rooms; the frontend converts socket events to window events, and any mounted composable can refetch its data slice.

Supports: Section 3.11.1 (Network Model), Section 4.5 (Objective No. 5 — Real-time monitoring)
server.js — Socket.io connection handler (rooms)
io.use(async (socket, next) => {

    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    socket.userId   = decoded.customerId || decoded.adminId || decoded.id;

  // Room: personal userId (CUST-xxxx / ADM-xxxx)

E.9 — Analytics Aggregation

Reports are computed entirely in MongoDB using aggregation pipelines — no data is loaded into Node to be counted. The getTopProducts pipeline below unwinds order items and groups by name, producing the Top Products chart in Figure 4.4.2.

Supports: Section 4.4 (Report Generation), Figure 4.4.2
services/AnalyticsServices.js — getTopProducts
async getTopProducts(limit = 5, dateFrom, dateTo) {
  const { start, end } = this._buildDateRange(dateFrom, dateTo);

const topProducts = await Order.aggregate([
  { $match: {
      isProvided: false,
      status:     'Completed',
      'items.name': { $exists: true, $ne: null },
      orderedAt:  { $gte: start, $lte: end },
  }},
  { $unwind: '$items' },                              // flatten items array
  { $group: {
      _id:           '$items.name',
      totalQuantity: { $sum: '$items.quantity' },
      totalRevenue:  { $sum: '$items.estimatedTotal' },
      orderCount:    { $sum: 1 },
  }},
  { $sort:  { totalQuantity: -1 } },               // best seller first
  { $limit: limit },
]);

  return { success: true, data: topProducts };
}

E.11 — Inventory and Stock Management

Products carry their own per-size stock in the Product document. Stock decrements happen inside the order-creation transaction (E.4) and are restored with a single atomic $inc on cancellation. This avoids the double-write pattern that would otherwise leave stock inconsistent.
